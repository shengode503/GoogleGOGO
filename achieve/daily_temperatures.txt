# Complexity: O(n), O(n)
def solve(temperatures):
    stack = []
    results = [0] * len(temperatures)
    for idx, temp in enumerate(temperatures):
        while stack and stack[-1][1] < temp:
            prev_idx, prev_temp = stack.pop()
            results[prev_idx] = idx - prev_idx
        stack.append((idx, temp))
    return results
