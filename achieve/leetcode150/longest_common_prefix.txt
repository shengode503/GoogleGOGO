# Complexity: O(mn), O(n)
def solve(strs):
    prefix = strs[0]
    for i in range(1, len(strs)):
        string = strs[i]
        new_prefix = []
        for j in range(min(len(prefix), len(string))):
            if prefix[j] == string[j]:
                new_prefix.append(string[j])
            else:
                break
        prefix = new_prefix
    return "".join(prefix)
