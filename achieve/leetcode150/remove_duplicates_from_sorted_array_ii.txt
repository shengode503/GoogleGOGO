# Complexity: O(n), O(1)
def solve(nums):
    count, replace_idx = 1, 1
    for i in range(1, len(nums)):
        if nums[i] == nums[i-1]:
            if count < 2:
                nums[replace_idx] = nums[i]
                replace_idx += 1
                count += 1
        else:
            count = 1
            nums[replace_idx] = nums[i]
            replace_idx += 1
    return replace_idx
