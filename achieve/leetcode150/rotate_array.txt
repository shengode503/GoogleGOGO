# Complexity: O(n), O(1)
def solve(nums, k):
    k %= len(nums)

    def rotate(start, end):
        while start <= end:
            nums[start], nums[end] = nums[end], nums[start]
            start += 1
            end -= 1

    rotate(0, len(nums) - 1)
    rotate(0, k - 1)
    rotate(k, len(nums) - 1)
    return nums
