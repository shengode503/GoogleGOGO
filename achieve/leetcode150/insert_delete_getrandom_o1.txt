# Complexity: O(1), (n)
import random

class RandomizedSet:
    def __init__(self):
        self.values =[]
        self.mapping = {}
    def insert(self, val):
        if val not in self.mapping:
            self.values.append(val)
            self.mapping[val] = len(self.values) - 1
            return True
        else:
            return False
        
    def remove(self, val):
        if val in self.mapping:
            idx = self.mapping[val]
            final_val = self.values[-1]

            self.values[idx], self.values[-1] = self.values[-1], self.values[idx]
            self.mapping[final_val] = idx

            self.values.pop()
            del self.mapping[val]       
    
            return True
        else:
            return False
    def getRandom(self):
        return self.values[random.randint(0, len(self.values) - 1)]
