# Complexity: O(n), O(1)
def solve(s):
    mapping = {
        "I": 1,
        "V": 5,
        "X": 10,
        "L": 50,
        "C": 100,
        "D": 500,
        "M": 1000
    }
    i  = 0
    result = 0
    prev_num = 0
    while i < len(s):
        curr_num = mapping[s[i]]
        if curr_num > prev_num:
           result += curr_num - prev_num - prev_num
        else:
            result += curr_num
        prev_num = curr_num
        i += 1
    return result
